# --- /etc/systemd/system/manic-api.service ---
[Unit]
Description=Manic AI Orchestrator API
After=network.target postgresql.service redis.service

[Service]
User=deploy
WorkingDirectory=/opt/manic-ai-orchestrator
EnvironmentFile=/opt/manic-ai-orchestrator/.env
ExecStart=/opt/manic-ai-orchestrator/venv/bin/uvicorn app.main:app --host 0.0.0.0 --port 8000
Restart=always
RestartSec=5

[Install]
WantedBy=multi-user.target


# --- /etc/systemd/system/manic-worker.service ---
[Unit]
Description=Manic AI Orchestrator Celery Worker
After=network.target postgresql.service redis.service

[Service]
User=deploy
WorkingDirectory=/opt/manic-ai-orchestrator
EnvironmentFile=/opt/manic-ai-orchestrator/.env
ExecStart=/opt/manic-ai-orchestrator/venv/bin/celery -A app.optimized_worker.celery_app worker --loglevel=info --concurrency=4
Restart=always
RestartSec=5

[Install]
WantedBy=multi-user.target
