# --- /etc/systemd/system/nexus-api.service ---
[Unit]
Description=Nexus Orchestrator API
After=network.target postgresql.service redis.service

[Service]
User=deploy
WorkingDirectory=/opt/nexus-orchestrator
EnvironmentFile=/opt/nexus-orchestrator/.env
ExecStart=/opt/nexus-orchestrator/venv/bin/uvicorn app.main:app --host 0.0.0.0 --port 8010
Restart=always
RestartSec=5

[Install]
WantedBy=multi-user.target


# --- /etc/systemd/system/nexus-worker.service ---
[Unit]
Description=Nexus Orchestrator Celery Worker
After=network.target postgresql.service redis.service

[Service]
User=deploy
WorkingDirectory=/opt/nexus-orchestrator
EnvironmentFile=/opt/nexus-orchestrator/.env
ExecStart=/opt/nexus-orchestrator/venv/bin/celery -A app.worker.celery_app worker --loglevel=info --concurrency=4
Restart=always
RestartSec=5

[Install]
WantedBy=multi-user.target
